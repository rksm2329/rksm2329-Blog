纪念自己磕出的第一道紫题。

## 题意
给定 $n, m$，求出：

$$
\sum _ {i = 1} ^ n \sum _ {i = 1} ^ m (n \bmod i) \times (m \bmod j)，i \ne j
$$

## 思路
没有 $i \ne j$，一切都好说。将式子拆开，直接数论分块求解即可。

但是多了这么一个条件。于是，我们考虑容斥。先计算出不包含 $i \ne j$ 的答案 $ans_0$，再减去 $i = j$ 的答案。

现在需要解决的问题是，如何求解 $i = j$ 时的答案。发现，当 $i \gt \min(n, m)$ 时，答案是 0，没有用。所以，我们只需做到 $\min(n, m)$ 即可。

将式子写下来：$\sum _ {i = 1} ^ {\min(n, m)} (n \bmod i) \times (m \bmod i)$

化为可用数论分块解决的形式：$\sum _ {i = 1} ^ {\min(n, m)} (n - \lfloor \frac n i \rfloor \cdot i) \times (m - \lfloor \frac m i \rfloor \cdot i)$

拆开：$\sum _ {i = 1} ^ {\min(n, m)} nm - i \cdot (n \cdot \lfloor \frac m i \rfloor + m \cdot \lfloor \frac n i \rfloor) + i ^ 2 \cdot \lfloor \frac n i \rfloor \lfloor \frac m i \rfloor$

其中，$nm$ 可以直接得到，$i \cdot (n \cdot \lfloor \frac m i \rfloor + m \cdot \lfloor \frac n i \rfloor) = i \cdot n \cdot \lfloor \frac m i \rfloor + i \cdot m \cdot \lfloor \frac n i \rfloor$ 可用数论分块计算，我们将其答案记为 $ans_1, ans_2$，$i ^ 2 \cdot \lfloor \frac n i \rfloor \lfloor \frac m i \rfloor$ 是个问题。

回顾数论分块的思想，将一个数分为根号个块，求解带有向下取整的问题。出现了两个数向下取整，但是我们仍然可以将其分为 2 倍根号个块，进行计算。每次将 $r$ 变为 $\min(\frac n {\lfloor \frac n l \rfloor}, \frac m {\lfloor \frac m l \rfloor})$，总共操作 2 倍根号次，将答案记为 $ans_3$。平方和如何计算？由前缀平方和公式：$\sum _ {i = 1} ^ n i ^ 2 = \frac {n (2n + 1) (n + 1)} 6$

将和拆分为 $[1, r], [1, l - 1]$ 分别求解即可。

不要忘了加上 $nm$。最后，答案为 $ans_0 - nm \cdot \min(n, m) + ans_1 + ans_2 - ans_3$。

#include <bits/extc++.h>

using namespace std;
using ll = long long;
using i128 = __int128;

namespace io {
	const int __SIZE = (1 << 21) + 1;
	char ibuf[__SIZE], *iS, *iT, obuf[__SIZE], *oS = obuf, *oT = oS + __SIZE - 1, __c, qu[55]; int __f, qr, _eof;
	#define Gc() (iS == iT ? (iT = (iS = ibuf) + fread (ibuf, 1, __SIZE, stdin), (iS == iT ? EOF : *iS ++)) : *iS ++)
	inline void flush () { fwrite (obuf, 1, oS - obuf, stdout), oS = obuf; }
	inline void gc (char &x) { x = Gc(); }
	inline void pc (char x) { *oS ++ = x; if (oS == oT) flush (); }
	inline void pstr (const char *s) { int __len = strlen(s); for (__f = 0; __f < __len; ++__f) pc (s[__f]); }
	inline void gstr (char *s) { for (__c = Gc(); __c < 32 || __c > 126 || __c == ' ';)  __c = Gc();
		for (; __c > 31 && __c < 127 && __c != ' ' && __c != '\n' && __c != '\r'; ++s, __c = Gc()) *s = __c; *s = 0; }
	template <class I> inline bool gi (I &x) { _eof = 0;
		for (__f = 1, __c = Gc(); (__c < '0' || __c > '9') && !_eof; __c = Gc()) { if (__c == '-') __f = -1; _eof |= __c == EOF; }
		for (x = 0; __c <= '9' && __c >= '0' && !_eof; __c = Gc()) x = x * 10 + (__c & 15), _eof |= __c == EOF; x *= __f; return !_eof; }
	template <class I> inline void print (I x) { if (!x) pc ('0'); if (x < 0) pc ('-'), x = -x;
		while (x) qu[++ qr] = x % 10 + '0', x /= 10; while (qr) pc (qu[qr --]); }
	struct Flusher_ {~Flusher_(){flush();}}io_flusher_;
} using io::pc; using io::gc; using io::pstr; using io::gstr; using io::gi; using io::print;

const int MOD = 19940417;

i128 n, m;

i128 Calc(i128 x) {
  return x * (x * 2 + 1) * (x + 1) / 6;
}

int main() {
  gi(n), gi(m);
  i128 ans1 = n * n, ans2 = m * m, ans3 = 0, ans4 = 0, ans5 = 0;
  for (ll l = 1, r; l <= n; l = r + 1) {
    r = min(n, n / (n / l));
    ans1 -= (l + r) * (r - l + 1) / 2 * (n / l);
  }
  for (ll l = 1, r; l <= m; l = r + 1) {
    r = min(m, m / (m / l));
    ans2 -= (l + r) * (r - l + 1) / 2 * (m / l);
  }
  for (ll l = 1, r; l <= min(n, m); l = r + 1) {
    r = min(n / (n / l), m / (m / l));
    ans3 += (Calc(r) - Calc(l - 1)) * (n / l) * (m / l);
  }
  for (i128 l = 1, r; l <= min(n, m); l = r + 1) {
    r = min(min(n, m), m / (m / l));
    ans4 += (l + r) * (r - l + 1) / 2 * (m / l);
  }
  ans4 *= n;
  for (i128 l = 1, r; l <= min(n, m); l = r + 1) {
    r = min(min(n, m), n / (n / l));
    ans5 += (l + r) * (r - l + 1) / 2 * (n / l);
  }
  ans5 *= m;
  ll ans = (ans1 * ans2 - m * n * min(m, n) + ans4 + ans5 - ans3) % MOD;
  print(ans);
  return 0;
}
